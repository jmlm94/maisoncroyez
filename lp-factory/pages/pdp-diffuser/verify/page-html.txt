    <script>
(function () {
  // Evita instalar el interceptor más de una vez si el script se carga 2 veces.
  if (window.__bundleCascadeInstalled) return;
  window.__bundleCascadeInstalled = true;

  const originalFetch = window.fetch;

  /* ─── Helpers ──────────────────────────────────────────────── */

  // Detecta si la request es una mutación del carrito (change/update).
  function isCartMutationUrl(url) {
    return typeof url === 'string' && /\/cart\/(change|update)(\.js)?(\?|$)/.test(url);
  }

  // Extrae los identificadores (line key o variant id) que se están
  // poniendo en cantidad 0 en esta request, sin importar si viene como
  // query string (?id=...&quantity=0) o como JSON body ({id, quantity})
  // o como batch update ({updates: {key: 0, ...}}).
  function getRemovalTargets(url, options) {
    const targets = [];

    try {
      const parsed = new URL(url, window.location.origin);
      if (parsed.pathname.indexOf('/cart/change') !== -1 &&
          parsed.searchParams.get('quantity') === '0' &&
          parsed.searchParams.has('id')) {
        targets.push(parsed.searchParams.get('id'));
      }
    } catch (e) { /* url no parseable como URL absoluta/relativa, ignorar */ }

    if (options && typeof options.body === 'string') {
      try {
        const body = JSON.parse(options.body);
        if (body.quantity === 0 && body.id) {
          targets.push(String(body.id));
        }
        if (body.updates && typeof body.updates === 'object') {
          Object.entries(body.updates).forEach(([key, qty]) => {
            if (qty === 0) targets.push(key);
          });
        }
      } catch (e) { /* body no era JSON, ignorar */ }
    }

    return targets;
  }

  // Lee el carrito actual y devuelve el _bundle_id de cada target que
  // sea efectivamente un producto padre de bundle (tiene _bundle_id
  // pero NO _bundle_component).
  async function findParentBundleIds(targetKeys) {
    const bundleIds = [];
    try {
      const res = await originalFetch('/cart.js', { credentials: 'include' });
      const cart = await res.json();

      cart.items.forEach(item => {
        const key = item.key;
        const variantIdStr = String(item.variant_id || item.id);
        const matches = targetKeys.includes(key) || targetKeys.includes(variantIdStr);
        if (!matches) return;

        const props = item.properties || {};
        const isParent = props._bundle_id && !props._bundle_component;
        if (isParent) bundleIds.push(props._bundle_id);
      });
    } catch (err) {
      console.error('[Bundle cascade] No se pudo leer /cart.js:', err);
    }
    return bundleIds;
  }

  // Elimina del carrito todas las fragancias hijas asociadas a los
  // bundleIds dados.
  async function cascadeRemoveChildren(bundleIds) {
    if (!bundleIds.length) return;

    try {
      const res = await originalFetch('/cart.js', { credentials: 'include' });
      const cart = await res.json();

      const updates = {};
      cart.items.forEach(item => {
        const props = item.properties || {};
        if (props._bundle_component && bundleIds.includes(props._bundle_id)) {
          updates[item.key] = 0;
        }
      });

      if (Object.keys(updates).length === 0) return;

      await originalFetch('/cart/update.js', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ updates })
      });

      document.dispatchEvent(new CustomEvent('cart:refresh'));
    } catch (err) {
      console.error('[Bundle cascade] Error eliminando fragancias asociadas:', err);
    }
  }

  /* ─── Interceptor de fetch ─────────────────────────────────── */
  window.fetch = async function (...args) {
    const [url, options] = args;

    let parentBundleIds = [];

    // Si la request va a poner en 0 algún line item, revisamos ANTES
    // de que se ejecute si ese item es un bundle padre (una vez que
    // se elimine, ya no podremos leer sus properties del carrito).
    if (isCartMutationUrl(url)) {
      const targets = getRemovalTargets(url, options);
      if (targets.length) {
        parentBundleIds = await findParentBundleIds(targets);
      }
    }

    const response = await originalFetch.apply(this, args);

    // Si efectivamente se removió un bundle padre y la request fue
    // exitosa, eliminamos en cascada sus fragancias asociadas.
    if (parentBundleIds.length && response.ok) {
      await cascadeRemoveChildren(parentBundleIds);
    }

    return response;
  };
})();
